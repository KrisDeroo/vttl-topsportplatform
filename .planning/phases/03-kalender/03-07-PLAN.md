    /**
     * DateTimePicker — Popover + shadcn Calendar + Input[type=time].
     * Phase 4/5-reusable compound. Used in Phase 3 EventCreate/EditSheet.
     *
     * Reference: src/components/players/player-create-form.tsx lines 312-358 (dateOfBirth pattern)

    'use client';

    import { CalendarIcon } from 'lucide-react';
    import { format } from 'date-fns';
    import { useState } from 'react';

    import { Button } from '@/components/ui/button';
    import { Calendar } from '@/components/ui/calendar';
    import { Input } from '@/components/ui/input';
    import { Popover, PopoverContent, PopoverTrigger } from '@/components/ui/popover';
    import { cn } from '@/lib/utils';

    interface Props {
      value: Date | undefined;
      onChange: (next: Date) => void;
      'aria-label'?: string;
    }

    export function DateTimePicker({ value, onChange, ...aria }: Props) {
      const [open, setOpen] = useState(false);
      const date = value ?? new Date();
      const time = format(date, 'HH:mm');

      function setDatePart(next: Date) {
        const merged = new Date(next);
        merged.setHours(date.getHours(), date.getMinutes(), 0, 0);
        onChange(merged);
        setOpen(false);
      }

      function setTimePart(value: string) {
        const [h, m] = value.split(':').map((n) => Number.parseInt(n, 10));
        const merged = new Date(date);
        merged.setHours(h || 0, m || 0, 0, 0);
        onChange(merged);
      }

      return (
        <div className="flex items-center gap-2">
          <Popover open={open} onOpenChange={setOpen}>
            <PopoverTrigger asChild>
              <Button
                variant="outline"
                className={cn('justify-start text-left font-normal', !value && 'text-muted-foreground')}
                {...aria}
              >
                <CalendarIcon className="mr-2 h-4 w-4" />
                {value ? format(date, 'dd/MM/yyyy') : '—'}
              </Button>
            </PopoverTrigger>
            <PopoverContent className="w-auto p-0" align="start">
              <Calendar mode="single" selected={value} onSelect={(d) => d && setDatePart(d)} initialFocus />
            </PopoverContent>
          </Popover>
          <Input
            type="time"
            value={time}
            onChange={(e) => setTimePart(e.target.value)}
            className="w-28"
            aria-label="Tijd"
          />
        </div>
      );
    }

    /**
     * RruleEditor — minimal RRULE v1 builder (Phase 3 scope).
     *
     * Per UI3-D12: ONLY "Deze afspraak" scope active in Phase 3; "Deze en toekomstige"
     * and "Alle afspraken in de reeks" disabled with "(Fase 4)" suffix in this widget.
     *
     * Builds an RFC 5545 string via RRule.optionsToString() (Pitfall 8 — never string-concat).
     * Emits via onChange(rruleString | undefined). Phase 4/5-reusable.
     *
     * Reference: 03-UI-SPEC.md §Form-field contract row "Recurring" line 323
     *            03-CONTEXT.md D-12 (UI3-D12)
     */
    'use client';

    import { addYears, format } from 'date-fns';
    import { useTranslations } from 'next-intl';
    import { useEffect, useState } from 'react';
    import { Frequency, RRule } from 'rrule';

    import { Input } from '@/components/ui/input';
    import { Label } from '@/components/ui/label';
    import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';

    type Freq = 'DAILY' | 'WEEKLY' | 'MONTHLY';
    type EndMode = 'never' | 'on_date' | 'after_n';

    interface Props {
      value: string | undefined;
      onChange: (rrule: string | undefined) => void;
      dtstart: Date;
    }

    const FREQ_MAP: Record<Freq, Frequency> = {
      DAILY: Frequency.DAILY,
      WEEKLY: Frequency.WEEKLY,
      MONTHLY: Frequency.MONTHLY,
    };

    export function RruleEditor({ value, onChange, dtstart }: Props) {
      const t = useTranslations('calendar.event.recurrence');
      const [freq, setFreq] = useState<Freq>('WEEKLY');
      const [interval, setInterval] = useState(1);
      const [endMode, setEndMode] = useState<EndMode>('never');
      const [endDate, setEndDate] = useState(format(addYears(dtstart, 1), 'yyyy-MM-dd'));
      const [endCount, setEndCount] = useState(10);

      useEffect(() => {
        const options: ConstructorParameters<typeof RRule>[0] = {
          freq: FREQ_MAP[freq],
          interval,
        };
        if (endMode === 'on_date') options.until = new Date(endDate);
        if (endMode === 'after_n') options.count = endCount;
        // 'never' is left to the server's ensureHorizon() to auto-inject UNTIL = +2y.
        try {
          const str = RRule.optionsToString({ ...options, dtstart: undefined });
          onChange(str);
        } catch {
          onChange(undefined);
        }
      }, [freq, interval, endMode, endDate, endCount, onChange]);

      return (
        <div className="space-y-3 rounded-md border border-border p-3">
          <div className="grid grid-cols-2 gap-3">
            <div>
              <Label>{t('frequency')}</Label>
              <Select value={freq} onValueChange={(v) => setFreq(v as Freq)}>
                <SelectTrigger><SelectValue /></SelectTrigger>
                <SelectContent>
                  <SelectItem value="DAILY">Dagelijks</SelectItem>
                  <SelectItem value="WEEKLY">Wekelijks</SelectItem>
                  <SelectItem value="MONTHLY">Maandelijks</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <div>
              <Label>{t('interval', { n: interval, unit: '' })}</Label>
              <Input
                type="number"
                min={1}
                max={4}
                value={interval}
                onChange={(e) => setInterval(Math.max(1, Math.min(4, Number(e.target.value))))}
              />
            </div>
          </div>
          <div>
            <Label>{t('endLabel')}</Label>
            <Select value={endMode} onValueChange={(v) => setEndMode(v as EndMode)}>
              <SelectTrigger><SelectValue /></SelectTrigger>
              <SelectContent>
                <SelectItem value="never">{t('endNever')}</SelectItem>
                <SelectItem value="on_date">{t('endOnDate')}</SelectItem>
                <SelectItem value="after_n">{t('endAfter', { n: endCount })}</SelectItem>
              </SelectContent>
            </Select>
          </div>
          {endMode === 'on_date' && (
            <Input type="date" value={endDate} onChange={(e) => setEndDate(e.target.value)} />
          )}
          {endMode === 'after_n' && (
            <Input type="number" min={1} value={endCount} onChange={(e) => setEndCount(Number(e.target.value))} />
          )}
        </div>
      );
    }

    /**
     * EventFilterBar — desktop inline + mobile bottom Sheet.
     * URL state in `?filter=<base64>` per UI3-D8.
     *
     * Desktop: 6 event-type chips + 4 FilterCombobox (player/trainer/sparring/academy)

     * controls stacked.
     *
     * Reference: 03-UI-SPEC.md §Filter Bar Contract lines 398-429
     */
    'use client';

    import { useTranslations } from 'next-intl';
    import { usePathname, useRouter, useSearchParams } from 'next/navigation';
    import { useCallback, useEffect, useState } from 'react';

    import { FilterCombobox } from '@/components/calendar/filter-combobox';
    import { Button } from '@/components/ui/button';
    import { Sheet, SheetContent, SheetHeader, SheetTitle } from '@/components/ui/sheet';
    import { ToggleGroup, ToggleGroupItem } from '@/components/ui/toggle-group';

    const EVENT_TYPES = [
      'event_type_training',
      'event_type_tournament',
      'event_type_meeting',
      'event_type_stage',
      'event_type_eval_conversation',
      'event_type_medical',
    ] as const;

    interface FilterState {
      types: string[];
      playerIds: string[];
      trainerIds: string[];
      sparringPartnerIds: string[];
      academies: string[];
    }

    function decodeFilter(raw: string | null): FilterState {
      const empty: FilterState = { types: [], playerIds: [], trainerIds: [], sparringPartnerIds: [], academies: [] };
      if (!raw) return empty;
      try {
        const parsed = JSON.parse(Buffer.from(raw, 'base64').toString());
        return { ...empty, ...parsed };
      } catch {
        return empty;
      }
    }

    function encodeFilter(state: FilterState): string | null {
      const nonEmpty = Object.fromEntries(
        Object.entries(state).filter(([_, v]) => Array.isArray(v) && v.length > 0),
      );
      if (Object.keys(nonEmpty).length === 0) return null;
      return Buffer.from(JSON.stringify(nonEmpty)).toString('base64');
    }

    export function EventFilterBar() {
      const t = useTranslations('calendar');
      const tLookup = useTranslations('lookup.eventType');
      const router = useRouter();
      const pathname = usePathname();
      const search = useSearchParams();
      const [mobileOpen, setMobileOpen] = useState(false);

      const initialFilter = decodeFilter(search.get('filter'));
      const [state, setState] = useState<FilterState>(initialFilter);

      // Listen for the toolbar's "Filters" button (mobile only).
      useEffect(() => {
        const handler = () => setMobileOpen(true);
        document.addEventListener('calendar:open-filters', handler);
        return () => document.removeEventListener('calendar:open-filters', handler);
      }, []);

      const pushFilter = useCallback(
        (next: FilterState) => {
          const params = new URLSearchParams(search);
          const encoded = encodeFilter(next);
          if (encoded) params.set('filter', encoded);
          else params.delete('filter');
          router.push(`${pathname}?${params.toString()}`);
        },
        [pathname, router, search],
      );

      const update = (patch: Partial<FilterState>) => {
        const next = { ...state, ...patch };
