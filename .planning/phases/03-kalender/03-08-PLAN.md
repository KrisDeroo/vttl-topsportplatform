    import { sql } from 'drizzle-orm';
    import type { drizzle } from 'drizzle-orm/postgres-js';
    import { v4 as uuidv4 } from 'uuid'; // already in repo via other deps

    export interface SeededCalendarFixtures {
      eventIds: {
        training: string;
        tournament: string;
        meeting: string;
        stage: string;
        evalConv: string;
        medical: string;
        trainingRecurring: string;
        overlappingA: string;
        overlappingB: string;
      };
      cancelledExceptionId: string;
    }

    /** Seed canonical calendar fixtures. Run AFTER seedRolesMatrix. */
    export async function seedCalendarFixtures(
      dbHandle: ReturnType<typeof drizzle>,
      seeded: { users: { technical_director: string; trainer: string; player: string; academy_manager: string; parent: string; sparring_partner: string; medical_staff: string } },
    ): Promise<SeededCalendarFixtures> {
      const TD = seeded.users.technical_director;
      const trainer = seeded.users.trainer;
      const player = seeded.users.player;
      const tNow = new Date();
      tNow.setHours(10, 0, 0, 0);
      const tEnd = new Date(tNow); tEnd.setHours(11, 30, 0, 0);

      const ids = {
        training: uuidv4(),
        tournament: uuidv4(),
        meeting: uuidv4(),
        stage: uuidv4(),
        evalConv: uuidv4(),
        medical: uuidv4(),
        trainingRecurring: uuidv4(),
        overlappingA: uuidv4(),
        overlappingB: uuidv4(),
      };
      const cancelledExceptionId = uuidv4();

      // Seed event_type lookup if not already (idempotent)
      await dbHandle.execute(sql`
        INSERT INTO event_type (code, sort_order, active) VALUES
          ('event_type_training', 1, true),
          ('event_type_tournament', 2, true),
          ('event_type_meeting', 3, true),
          ('event_type_stage', 4, true),
          ('event_type_eval_conversation', 5, true),
          ('event_type_medical', 6, true)
        ON CONFLICT (code) DO NOTHING
      `);

      // 1. Single training
      await dbHandle.execute(sql`
        INSERT INTO calendar_events (id, type_code, title, starts_at, ends_at, all_day, location, created_by)
        VALUES (${ids.training}::uuid, 'event_type_training', 'Senioren A training',
                ${tNow.toISOString()}, ${tEnd.toISOString()}, false, 'Topsportschool', ${TD}::uuid)
      `);
      await dbHandle.execute(sql`
        INSERT INTO training_sessions (event_id, duration_minutes, training_type_code, organisation_code, trainer_id)
        VALUES (${ids.training}::uuid, 90, 'training_type_group', 'org_academy', ${trainer}::uuid)
      `);
      await dbHandle.execute(sql`
        INSERT INTO calendar_event_participants (event_id, user_id, role_in_event)
        VALUES (${ids.training}::uuid, ${player}::uuid, 'participant')
      `);

      // 2. Tournament
      const tomorrow = new Date(tNow); tomorrow.setDate(tomorrow.getDate() + 1);
      const tomorrowEnd = new Date(tomorrow); tomorrowEnd.setHours(18, 0, 0, 0);
      await dbHandle.execute(sql`
        INSERT INTO calendar_events (id, type_code, title, starts_at, ends_at, all_day, created_by)
        VALUES (${ids.tournament}::uuid, 'event_type_tournament', 'Belgian Open', ${tomorrow.toISOString()}, ${tomorrowEnd.toISOString()}, false, ${TD}::uuid)
      `);
      await dbHandle.execute(sql`
        INSERT INTO tournaments (event_id, city, country, age_category_code, tournament_type_code)
        VALUES (${ids.tournament}::uuid, 'Antwerpen', 'BE', 'age_senior', 'tournament_belgium')
      `);

      // 3. Meeting / 4. Stage / 5. EvalConv / 6. Medical — minimal
      const dayAfter = new Date(tNow); dayAfter.setDate(dayAfter.getDate() + 2);
      const dayAfterEnd = new Date(dayAfter); dayAfterEnd.setHours(dayAfter.getHours() + 1, 0, 0, 0);
      await dbHandle.execute(sql`
        INSERT INTO calendar_events (id, type_code, title, starts_at, ends_at, created_by)
        VALUES
          (${ids.meeting}::uuid, 'event_type_meeting', 'Coaches stafvergadering',
           ${dayAfter.toISOString()}, ${dayAfterEnd.toISOString()}, ${TD}::uuid),
          (${ids.stage}::uuid, 'event_type_stage', 'Zomerstage Spanje',
           ${dayAfter.toISOString()}, ${dayAfterEnd.toISOString()}, ${TD}::uuid),
          (${ids.evalConv}::uuid, 'event_type_eval_conversation', 'Halfjaarlijks gesprek',
           ${dayAfter.toISOString()}, ${dayAfterEnd.toISOString()}, ${TD}::uuid),
          (${ids.medical}::uuid, 'event_type_medical', 'Knie-controle',
           ${dayAfter.toISOString()}, ${dayAfterEnd.toISOString()}, ${TD}::uuid)
      `);
      await dbHandle.execute(sql`INSERT INTO meetings (event_id) VALUES (${ids.meeting}::uuid)`);
      await dbHandle.execute(sql`
        INSERT INTO stages (event_id, place, country) VALUES (${ids.stage}::uuid, 'Madrid', 'ES')
      `);
      await dbHandle.execute(sql`
        INSERT INTO eval_conversations (event_id, evaluator_user_id, player_user_id)
        VALUES (${ids.evalConv}::uuid, ${trainer}::uuid, ${player}::uuid)
      `);
      await dbHandle.execute(sql`
        INSERT INTO medical_appointments (event_id, is_injury, doctor)
        VALUES (${ids.medical}::uuid, false, 'Dr. Janssens')
      `);

      // 7. Recurring training (weekly Tuesday 10:00 with UNTIL +1y)
      const oneYearOut = new Date(tNow); oneYearOut.setFullYear(oneYearOut.getFullYear() + 1);
      const rrule = `RRULE:FREQ=WEEKLY;UNTIL=${oneYearOut.toISOString().replace(/[-:]|\.\d{3}/g, '')};BYDAY=TU`;
      await dbHandle.execute(sql`
        INSERT INTO calendar_events (id, type_code, title, starts_at, ends_at, rrule, created_by)
        VALUES (${ids.trainingRecurring}::uuid, 'event_type_training', 'Wekelijkse jeugd-training',
                ${tNow.toISOString()}, ${tEnd.toISOString()}, ${rrule}, ${TD}::uuid)
      `);
      await dbHandle.execute(sql`
        INSERT INTO training_sessions (event_id, duration_minutes, training_type_code, organisation_code, trainer_id)
        VALUES (${ids.trainingRecurring}::uuid, 90, 'training_type_group', 'org_academy', ${trainer}::uuid)
      `);

      // 8. Cancelled exception on the recurring
      const nextTuesday = new Date(tNow);
      while (nextTuesday.getDay() !== 2) nextTuesday.setDate(nextTuesday.getDate() + 1);
      await dbHandle.execute(sql`
        INSERT INTO calendar_event_exceptions (id, event_id, occurrence_date, cancelled, created_by)
        VALUES (${cancelledExceptionId}::uuid, ${ids.trainingRecurring}::uuid,
                ${nextTuesday.toISOString().slice(0, 10)}::date, true, ${TD}::uuid)
      `);

      // 9. Overlapping pair (same participant, overlapping times) for conflict tests
      const overlap = new Date(tNow); overlap.setDate(overlap.getDate() + 3);
      const overlapEnd = new Date(overlap); overlapEnd.setHours(overlap.getHours() + 2, 0, 0, 0);
      const overlapPlus = new Date(overlap); overlapPlus.setHours(overlap.getHours() + 1, 0, 0, 0);
      const overlapPlusEnd = new Date(overlapPlus); overlapPlusEnd.setHours(overlapPlus.getHours() + 1, 0, 0, 0);
      await dbHandle.execute(sql`
        INSERT INTO calendar_events (id, type_code, title, starts_at, ends_at, created_by) VALUES
          (${ids.overlappingA}::uuid, 'event_type_meeting', 'Meeting A',
           ${overlap.toISOString()}, ${overlapEnd.toISOString()}, ${TD}::uuid),
          (${ids.overlappingB}::uuid, 'event_type_meeting', 'Meeting B',
           ${overlapPlus.toISOString()}, ${overlapPlusEnd.toISOString()}, ${TD}::uuid)
      `);
      await dbHandle.execute(sql`INSERT INTO meetings (event_id) VALUES (${ids.overlappingA}::uuid)`);
      await dbHandle.execute(sql`INSERT INTO meetings (event_id) VALUES (${ids.overlappingB}::uuid)`);
      await dbHandle.execute(sql`
        INSERT INTO calendar_event_participants (event_id, user_id, role_in_event) VALUES
          (${ids.overlappingA}::uuid, ${player}::uuid, 'participant'),
          (${ids.overlappingB}::uuid, ${player}::uuid, 'participant')
      `);

      return { eventIds: ids, cancelledExceptionId };
    }
