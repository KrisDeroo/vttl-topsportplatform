    ```typescript
    import { z } from 'zod';

    /**
     * Phase 4 ranking Zod schemas — split-column discriminated union (D-86).
     *
     * value is a discriminated union over 'kind':
     *   - 'numeric' → value.value is a positive number (international ranks)
     *   - 'classification' → value.code is a belgium_classification.code FK
     *
     * The router cross-checks input.value.kind against ranking_type.value_shape
     * (also stored in DB) — mismatch returns BAD_REQUEST with i18n key.
     */

    const numericValueBranch = z.object({
      kind: z.literal('numeric'),
      value: z.number().positive({ message: 'errors.ranking.mustBePositive' }),
    }).strict();

    const classificationValueBranch = z.object({
      kind: z.literal('classification'),
      code: z.string().min(1, { message: 'errors.ranking.classificationRequired' }),
    }).strict();

    export const addEntryInput = z.object({
      playerUserId: z.string().uuid(),
      rankingTypeCode: z.string().min(1),
      recordedAt: z.coerce.date({ errorMap: () => ({ message: 'errors.field.invalidDate' }) }),
      source: z.enum(['manual', 'federation_official']).default('manual'),
      value: z.discriminatedUnion('kind', [numericValueBranch, classificationValueBranch]),
      _meta: z.object({ idempotencyKey: z.string().min(8).max(128).optional() }).optional(),
    }).strict();

    export const getHistoryInput = z.object({
      playerUserId: z.string().uuid(),
      rankingTypeCode: z.string().min(1),
      from: z.coerce.date().optional(),
      to: z.coerce.date().optional(),
    }).strict();

    export const getCurrentByTypeInput = z.object({
      playerUserId: z.string().uuid(),
      rankingTypeCode: z.string().min(1),
    }).strict();

    export const listEntriesInput = z.object({
      playerUserId: z.string().uuid(),
      rankingTypeCode: z.string().min(1).optional(),
    }).strict();
    ```

    **Step 2 — `src/server/trpc/routers/ranking.ts`:**

    Implementation outline (executor expands inline using full code shape from RESEARCH §Pattern 4):

    ```typescript
    /**
     * ranking router — Phase 4 (D-86..D-91 + RANK-01..07 + DOM-RANK-01).
     *
     * Procedures:
     *   - addEntry — player or TD only (D-89). Composed with idempotency
     *     middleware (VALID-08). Cross-checks value.kind against
     *     ranking_type.value_shape. Atomic single-row insert + audit.
     *   - getHistory — player + type + date range; RLS does scope.
     *   - getCurrentByType — LIMIT 1 ORDER BY recorded_at DESC (RANK-05).
     *   - listEntries — audit/correction view (Rankings tab secondary tab).
     */
    import { TRPCError } from '@trpc/server';
    import { and, asc, desc, eq, gte, lte } from 'drizzle-orm';

    import { rankingEntries, rankingType } from '@/server/db/schema';

    import { router } from '../trpc';
    import { writeAudit } from '../middleware/audit';
    import { idempotencyMiddleware } from '../middleware/idempotency';
    import { protectedProcedure } from '../middleware/freshSession';
    import {
      addEntryInput,
      getHistoryInput,
      getCurrentByTypeInput,
      listEntriesInput,
    } from '../schemas/ranking';

    export const rankingRouter = router({

      addEntry: protectedProcedure
        .use(idempotencyMiddleware('ranking.addEntry'))
        .input(addEntryInput)
        .mutation(async ({ ctx, input }) => {
          // 1. RBAC: D-89 — player or TD; trainer rejected.
          if (ctx.scope.role !== 'player' && ctx.scope.role !== 'technical_director') {
            throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.role.notAllowed' });
          }
          if (ctx.scope.role === 'player' && input.playerUserId !== ctx.scope.userId) {
            throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.ranking.notOwnPlayer' });
          }

          // 2. Cross-check value.kind against ranking_type.value_shape (RESEARCH §Pattern 4)
          const rt = await ctx.db.select({
            valueShape: rankingType.valueShape,
            active: rankingType.active,
          }).from(rankingType).where(eq(rankingType.code, input.rankingTypeCode)).limit(1);

          if (!rt[0]) {
            throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.ranking.unknownType' });
          }
          if (!rt[0].active) {
            throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.ranking.inactiveType' });
          }
          if (rt[0].valueShape === 'numeric' && input.value.kind !== 'numeric') {
            throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.ranking.expectedNumeric' });
          }
          if (rt[0].valueShape === 'classification' && input.value.kind !== 'classification') {
            throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.ranking.expectedClassification' });
          }

          // 3. Insert (DB-level CHECK XOR provides defense in depth)
          const [inserted] = await ctx.db.insert(rankingEntries).values({
            playerUserId: input.playerUserId,
            rankingTypeCode: input.rankingTypeCode,
            recordedAt: input.recordedAt,
            source: input.source,
            valueNumeric: input.value.kind === 'numeric' ? input.value.value.toString() : null,
            valueClassificationCode: input.value.kind === 'classification' ? input.value.code : null,
            enteredBy: ctx.scope.userId,
            enteredAt: new Date(),
          }).returning({ id: rankingEntries.id });

          // 4. Audit
          await writeAudit(ctx, {
            action: 'ranking_entry_added',
            resourceType: 'ranking_entry',
            resourceId: inserted.id,
            newValues: {
              playerUserId: input.playerUserId,
              rankingTypeCode: input.rankingTypeCode,
              source: input.source,
              valueKind: input.value.kind,
              value: input.value.kind === 'numeric' ? input.value.value : input.value.code,
            },
            outcome: 'success',
          });

          return { ok: true as const, id: inserted.id };
        }),

      getHistory: protectedProcedure
        .input(getHistoryInput)
        .query(async ({ ctx, input }) => {
          const conditions = [
            eq(rankingEntries.playerUserId, input.playerUserId),
            eq(rankingEntries.rankingTypeCode, input.rankingTypeCode),
          ];
          if (input.from) conditions.push(gte(rankingEntries.recordedAt, input.from));
          if (input.to) conditions.push(lte(rankingEntries.recordedAt, input.to));

          // RLS via ranking_entry_visible_to helper (Plan 04-02).
          const rows = await ctx.db.select({
            id: rankingEntries.id,
            recordedAt: rankingEntries.recordedAt,
            source: rankingEntries.source,
            valueNumeric: rankingEntries.valueNumeric,
            valueClassificationCode: rankingEntries.valueClassificationCode,
            enteredBy: rankingEntries.enteredBy,
            enteredAt: rankingEntries.enteredAt,
          }).from(rankingEntries).where(and(...conditions)).orderBy(asc(rankingEntries.recordedAt));

          return { entries: rows };
        }),

      getCurrentByType: protectedProcedure
        .input(getCurrentByTypeInput)
        .query(async ({ ctx, input }) => {
          // RANK-05: latest entry only, never a flat field on player
          const rows = await ctx.db.select().from(rankingEntries)
            .where(and(
              eq(rankingEntries.playerUserId, input.playerUserId),
              eq(rankingEntries.rankingTypeCode, input.rankingTypeCode),
            ))
            .orderBy(desc(rankingEntries.recordedAt))
            .limit(1);
          return { current: rows[0] ?? null };
        }),

      listEntries: protectedProcedure
        .input(listEntriesInput)
        .query(async ({ ctx, input }) => {
          const conditions = [eq(rankingEntries.playerUserId, input.playerUserId)];
          if (input.rankingTypeCode) {
            conditions.push(eq(rankingEntries.rankingTypeCode, input.rankingTypeCode));
          }
          const rows = await ctx.db.select().from(rankingEntries)
            .where(and(...conditions))
            .orderBy(desc(rankingEntries.recordedAt));
          return { entries: rows };
        }),
    });
    ```

    Adapt field names to exact Drizzle column types (e.g., `valueNumeric` as a numeric column — drizzle treats numeric as string by default; the `.toString()` keeps the input compatible with the DB column type).
