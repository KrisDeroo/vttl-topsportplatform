    **Step 1 — Add `editRecurring` to calendar router:**

    Implementation outline (full tx body from RESEARCH §"Migration of participants"):

    ```typescript
    event: {
      // ... existing procedures (create, update, delete, declineParticipation, cancelOccurrence, get, detectConflicts) ...

      editRecurring: protectedProcedure
        .input(editRecurringInput)
        .mutation(async ({ ctx, input }) => {
          // 1. Load old event + RLS scope check (NOT_FOUND for out-of-scope)
          const oldEvent = await ctx.db.select().from(calendarEvents)
            .where(eq(calendarEvents.id, input.eventId)).limit(1);
          if (!oldEvent[0]) throw new TRPCError({ code: 'NOT_FOUND' });
          if (!oldEvent[0].rrule) {
            throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.calendar.notRecurring' });
          }

          // 2. RBAC: per-event-type create gate (defense in depth per Phase 3 D-48)
          //    Reuse canCreateEventType helper from middleware/calendarCreate.ts.
          if (!canCreateEventType(ctx.scope.role, oldEvent[0].typeCode)) {
            throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.calendar.roleCannotEditType' });
          }

          // 3. Dispatch by scope
          if (input.scope === 'single') {
            // Delegate to Phase 3 cancelOccurrence / exception-write flow (D-54).
            // This is a thin wrapper around the existing exception INSERT.
            // (Existing handler shipped Phase 3; here we just route to it.)
            return await applyExceptionForSingle(ctx, input);
          }

          if (input.scope === 'this_and_future') {
            // RESEARCH §"Migration of participants" — full tx body
            const split = splitRRule(oldEvent[0].rrule, input.splitDate!, oldEvent[0].startsAt);

            const result = await ctx.db.transaction(async (tx) => {
              // 3a. UPDATE old event with truncated RRULE
              await tx.update(calendarEvents)
                .set({ rrule: split.oldRruleString, updatedAt: new Date() })
                .where(eq(calendarEvents.id, oldEvent[0].id));

              // 3b. INSERT new event with edits applied + continuation RRULE
              const newDuration = oldEvent[0].endsAt.getTime() - oldEvent[0].startsAt.getTime();
              const newStartsAt = input.edits.startsAt ?? split.newDtstart;
              const newEndsAt = input.edits.endsAt ?? new Date(newStartsAt.getTime() + newDuration);
              const newRruleWithHorizon = ensureHorizon(split.newRruleString, newStartsAt);

              const [newEvent] = await tx.insert(calendarEvents).values({
                typeCode: oldEvent[0].typeCode,
                title: input.edits.title ?? oldEvent[0].title,
                startsAt: newStartsAt,
                endsAt: newEndsAt,
                allDay: oldEvent[0].allDay,
                location: input.edits.location !== undefined ? input.edits.location : oldEvent[0].location,
                description: input.edits.description !== undefined ? input.edits.description : oldEvent[0].description,
                rrule: newRruleWithHorizon,
                createdBy: ctx.scope.userId,
                createdAt: new Date(),
                updatedAt: new Date(),
              }).returning({ id: calendarEvents.id });

              // 3c. Copy extension row (training_sessions / meetings) with edited fields
              await copyExtensionRow(tx, oldEvent[0].typeCode, oldEvent[0].id, newEvent.id, input.edits);

              // 3d. Copy calendar_event_participants (series-level), RSVPs reset to pending
              const oldParts = await tx.select().from(calendarEventParticipants)
                .where(eq(calendarEventParticipants.eventId, oldEvent[0].id));
              if (oldParts.length > 0) {
                await tx.insert(calendarEventParticipants).values(
                  oldParts.map(p => ({
                    eventId: newEvent.id,
                    userId: p.userId,
                    roleInEvent: p.roleInEvent,
                    rsvpStatus: 'pending',
                    createdAt: new Date(),
                  })),
                );
              }

              // 3e. Copy session_sparring_partners if training_sessions (TRAIN-06)
              if (oldEvent[0].typeCode === 'event_type_training') {
                const oldSparring = await tx.select().from(sessionSparringPartners)
                  .where(eq(sessionSparringPartners.eventId, oldEvent[0].id));
                if (oldSparring.length > 0) {
                  await tx.insert(sessionSparringPartners).values(
                    oldSparring.map(s => ({
                      eventId: newEvent.id,
                      sparringPartnerId: s.sparringPartnerId,
                      createdBy: ctx.scope.userId,
                      createdAt: new Date(),
                    })),
                  );
                }
              }

              // 3f. NOTE: session_participants rows STAY on old event (D-83 immutable past).
              //     Explicitly DO NOT touch session_participants.

              return { newEventId: newEvent.id };
            });

            // 4. Audit
            await writeAudit(ctx, {
              action: 'calendar_event_recurring_split',
              resourceType: 'calendar_event',
              resourceId: oldEvent[0].id,
              oldValues: oldEvent[0],
              newValues: { newEventId: result.newEventId, splitDate: input.splitDate!.toISOString() },
              outcome: 'success',
            });

            return { ok: true as const, scope: 'this_and_future' as const, newEventId: result.newEventId };
          }

          if (input.scope === 'all_in_series') {
            // UPDATE base event + extension fields in place.
            // Past session_participants UNTOUCHED (D-83 — UPDATE on calendar_events does not cascade to session_participants).
            // Future exceptions whose occurrence_date no longer matches the new expansion BECOME INERT (zombie kept per RESEARCH §Open Question 2 + UI4-D20).

            await ctx.db.transaction(async (tx) => {
              const oldRrule = oldEvent[0].rrule!;
              let newRrule = oldRrule;
              if (input.edits.byday || input.edits.frequency) {
                // Rebuild RRULE preserving DTSTART
                const parsed = parseRrule(oldRrule, oldEvent[0].startsAt);
                const opts = { ...parsed.origOptions };
                if (input.edits.frequency) {
                  opts.freq = input.edits.frequency === 'daily' ? RRule.DAILY
                            : input.edits.frequency === 'weekly' ? RRule.WEEKLY
                            : RRule.MONTHLY;
                }
                if (input.edits.byday) {
                  opts.byweekday = input.edits.byday.map(d => RRule[d as 'MO' | 'TU' | 'WE' | 'TH' | 'FR' | 'SA' | 'SU']);
                }
                const { dtstart: _strip, ...rest } = opts;
                newRrule = RRule.optionsToString(rest);
              }

              await tx.update(calendarEvents).set({
                title: input.edits.title ?? oldEvent[0].title,
                startsAt: input.edits.startsAt ?? oldEvent[0].startsAt,
                endsAt: input.edits.endsAt ?? oldEvent[0].endsAt,
                location: input.edits.location !== undefined ? input.edits.location : oldEvent[0].location,
                description: input.edits.description !== undefined ? input.edits.description : oldEvent[0].description,
                rrule: newRrule,
                updatedAt: new Date(),
              }).where(eq(calendarEvents.id, oldEvent[0].id));

              // Extension-row update via helper
              await updateExtensionRow(tx, oldEvent[0].typeCode, oldEvent[0].id, input.edits);
            });

            // 4. Audit
            await writeAudit(ctx, {
              action: 'calendar_event_recurring_updated_all',
              resourceType: 'calendar_event',
              resourceId: oldEvent[0].id,
              oldValues: oldEvent[0],
              newValues: input.edits,
              outcome: 'success',
            });

            return { ok: true as const, scope: 'all_in_series' as const };
          }

          throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.calendar.unknownScope' });
        }),
    }
    ```

    ```typescript
    // calendar.event.attachSparringPartners — TD only (D-79 + D-63)
    attachSparringPartners: tdProcedure
      .input(z.object({
        eventId: z.string().uuid(),
        sparringPartnerIds: z.array(z.string().uuid()).min(1),
      }).strict())
      .mutation(async ({ ctx, input }) => {
        // 1. Verify each id is a user with role='sparring_partner' (app-layer FK row filter per Assumption A5)
        const rows = await ctx.db.select({ id: users.id, role: users.role }).from(users)
          .where(inArray(users.id, input.sparringPartnerIds));
        const invalid = rows.filter(r => r.role !== 'sparring_partner').map(r => r.id);
        if (rows.length !== input.sparringPartnerIds.length || invalid.length > 0) {
          throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.sparring.notASparringPartner' });
        }

        // 2. Insert junction rows (RLS in 0018 already enforces TD-only at DB)
        await ctx.db.insert(sessionSparringPartners).values(
          input.sparringPartnerIds.map(spId => ({
            eventId: input.eventId,
            sparringPartnerId: spId,
            createdBy: ctx.scope.userId,
            createdAt: new Date(),
          })),
        ).onConflictDoNothing();

        // 3. Audit
        for (const spId of input.sparringPartnerIds) {
          await writeAudit(ctx, {
            action: 'sparring_partner_attached',
            resourceType: 'session_sparring_partner',
            resourceId: `${input.eventId}:${spId}`,
            newValues: { eventId: input.eventId, sparringPartnerId: spId },
          });
        }
        return { ok: true as const, attached: input.sparringPartnerIds.length };
      }),
    ```
