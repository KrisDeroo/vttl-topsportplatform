    **Step 1 — `src/components/common/star-rating-input.tsx`:**

    ```typescript
    'use client';
    import { Star } from 'lucide-react';
    import { useTranslations } from 'next-intl';
    import { useId } from 'react';

    import { cn } from '@/lib/utils';
    import { mapStarsToDb, mapDbToStars } from '@/lib/quality-score';

    interface StarRatingInputProps {
      value: number | null;          // 1..10 from DB
      onChange: (next: number | null) => void;
      disabled?: boolean;
      ariaLabelledById?: string;
      name?: string;
    }

    export function StarRatingInput({ value, onChange, disabled, ariaLabelledById, name }: StarRatingInputProps) {
      const t = useTranslations('training.score.stars');
      const groupId = useId();
      const stars = mapDbToStars(value);

      const onSelect = (n: 0 | 1 | 2 | 3 | 4 | 5) => {
        if (disabled) return;
        if (n === stars) {
          onChange(null); // clear by clicking already-filled star (UI4-D05)
          return;
        }
        onChange(mapStarsToDb(n));
      };

      const onKey = (e: React.KeyboardEvent) => {
        if (disabled) return;
        if (['1', '2', '3', '4', '5'].includes(e.key)) {
          e.preventDefault();
          onSelect(Number(e.key) as 1 | 2 | 3 | 4 | 5);
        } else if (e.key === '0' || e.key === 'Escape') {
          e.preventDefault();
          onSelect(0);
        } else if (e.key === 'ArrowRight') {
          e.preventDefault();
          onSelect(Math.min(stars + 1, 5) as 1 | 2 | 3 | 4 | 5);
        } else if (e.key === 'ArrowLeft') {
          e.preventDefault();
          onSelect(Math.max(stars - 1, 0) as 0 | 1 | 2 | 3 | 4);
        }
      };

      return (
        <div
          id={groupId}
          role="radiogroup"
          aria-labelledby={ariaLabelledById}
          className="inline-flex gap-0.5"
          onKeyDown={onKey}
          tabIndex={0}
          data-testid="star-rating-input"
        >
          {[1, 2, 3, 4, 5].map((n) => (
            <button
              key={n}
              type="button"
              role="radio"
              aria-checked={stars === n}
              aria-label={t('aria', { n })}
              disabled={disabled}
              onClick={() => onSelect(n as 1 | 2 | 3 | 4 | 5)}
              className={cn(
                'p-0.5 transition-colors focus-visible:outline-2 focus-visible:outline-offset-2',
                disabled && 'cursor-not-allowed opacity-50',
              )}
            >
              <Star
                size={28}
                className={cn(
                  'transition-colors',
                  n <= stars ? 'fill-current text-[var(--cal-event-evalconv-border)]' : 'text-muted-foreground',
                )}
                aria-hidden
              />
            </button>
          ))}
          {stars > 0 && !disabled && (
            <button
              type="button"
              onClick={() => onSelect(0)}
              className="ml-2 text-xs text-muted-foreground hover:underline"
              aria-label={t('clear')}
            >
              {t('clear')}
            </button>
          )}
        </div>
      );
    }
    ```
