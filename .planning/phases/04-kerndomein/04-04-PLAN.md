    Implementation skeleton (full code shape from RESEARCH §Pattern 3 lines 626-718 — copy verbatim, adapting variable names to project conventions):

    Procedure composition:
    ```typescript
    enterResult: protectedProcedure
      .use(idempotencyMiddleware('tournament.enterResult'))
      .input(enterResultInput)
      .mutation(async ({ ctx, input }) => {
        // 1. Empty matches[] rejected by Zod .min(1) already; defense-in-depth check
        if (input.matches.length === 0) {
          throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.tournament.atLeastOneMatchRequired' });
        }

        // 2. Load tournament event for endsAt (wall) + startsAt (age-category snapshot)
        const event = await ctx.db.select({
          startsAt: calendarEvents.startsAt,
          endsAt: calendarEvents.endsAt,
          typeCode: calendarEvents.typeCode,
        }).from(calendarEvents).where(eq(calendarEvents.id, input.tournamentEventId)).limit(1);

        if (!event[0]) throw new TRPCError({ code: 'NOT_FOUND' });
        if (event[0].typeCode !== 'event_type_tournament') {
          throw new TRPCError({ code: 'BAD_REQUEST', message: 'errors.tournament.notATournament' });
        }

        // 3. D-71 player 14d wall; D-73 trainer/TD bypass
        const FOURTEEN_DAYS_MS = 14 * 24 * 60 * 60 * 1000;
        const wallExpired = Date.now() - event[0].endsAt.getTime() > FOURTEEN_DAYS_MS;
        if (ctx.scope.role === 'player' && wallExpired) {
          await writeAudit(ctx, {
            action: 'tournament_entry_window_expired_attempt',
            resourceType: 'tournament',
            resourceId: input.tournamentEventId,
            outcome: 'denied',
          });
          throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.tournament.entryWindowExpired' });
        }

        // 4. RBAC: player can only enter own; trainer must share academy with player; TD anytime
        if (ctx.scope.role === 'player' && input.playerUserId !== ctx.scope.userId) {
          throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.tournament.notOwnPlayer' });
        }
        if (ctx.scope.role === 'trainer') {
          const sharedAcademy = await ctx.db.execute(sql`
            SELECT 1 FROM academy_memberships am_player
            JOIN academy_memberships am_caller ON am_caller.academy_code = am_player.academy_code
            WHERE am_player.user_id = ${input.playerUserId} AND am_player.role = 'player'
              AND am_caller.user_id = ${ctx.scope.userId} AND am_caller.role = 'trainer'
            LIMIT 1
          `);
          if (sharedAcademy.length === 0) {
            throw new TRPCError({ code: 'FORBIDDEN', message: 'errors.tournament.trainerNotInAcademy' });
          }
        }

        // 5. entered_by attribution
        const enteredBy: 'player' | 'trainer' | 'td' =
          ctx.scope.role === 'player' ? 'player'
          : ctx.scope.role === 'technical_director' ? 'td'
          : 'trainer';

        // 6. DOM-CAT-02: snapshot age category at tournament start
        const cat = await getAgeCategoryAt(input.playerUserId, event[0].startsAt);
        const playerAgeCategoryCode = cat?.code ?? 'age_unknown';

        // 7. Snapshot pre-state for audit (TD overwrite path per D-75)
        const preTournamentRow = await ctx.db.select().from(tournamentResults)
          .where(and(
            eq(tournamentResults.tournamentEventId, input.tournamentEventId),
            eq(tournamentResults.playerUserId, input.playerUserId),
          )).limit(1);
        const preMatchRows = await ctx.db.select().from(matchResults)
          .where(and(
            eq(matchResults.tournamentEventId, input.tournamentEventId),
            eq(matchResults.playerUserId, input.playerUserId),
          ));
        const isOverwrite = preTournamentRow.length > 0;

        // 8. Atomic tx — UPSERT tournament_results + DELETE+INSERT match_results
        await ctx.db.transaction(async (tx) => {
          await tx.insert(tournamentResults).values({
            tournamentEventId: input.tournamentEventId,
            playerUserId: input.playerUserId,
            outcomeLevelCode: input.outcome,
            playerAgeCategoryCode,
            enteredBy,
            enteredByUserId: ctx.scope.userId,
            enteredAt: new Date(),
            updatedAt: new Date(),
          }).onConflictDoUpdate({
            target: [tournamentResults.tournamentEventId, tournamentResults.playerUserId],
            set: {
              outcomeLevelCode: input.outcome,
              enteredBy,
              enteredByUserId: ctx.scope.userId,
              updatedAt: new Date(),
            },
          });

          // Full replacement: DELETE then INSERT (RESEARCH Pattern 3 + Open Question 3)
          await tx.delete(matchResults).where(and(
            eq(matchResults.tournamentEventId, input.tournamentEventId),
            eq(matchResults.playerUserId, input.playerUserId),
          ));

          await tx.insert(matchResults).values(input.matches.map(m => ({
            tournamentEventId: input.tournamentEventId,
            playerUserId: input.playerUserId,
            roundCode: m.round,
            opponentName: m.opponent,
            opponentRanking: m.opponentRanking,
            matchDate: m.matchDate.toISOString().slice(0, 10),
            setsWon: m.setsWon,
            setsLost: m.setsLost,
            videoLink: m.videoLink,
          })));
        });

        // 9. Audit
        await writeAudit(ctx, {
          action: isOverwrite && ctx.scope.role === 'technical_director' ? 'tournament_result_overwritten' : 'tournament_result_entered',
          resourceType: 'tournament_result',
          resourceId: input.tournamentEventId + ':' + input.playerUserId,
          oldValues: isOverwrite ? { tournament: preTournamentRow[0], matches: preMatchRows } : undefined,
          newValues: { outcome: input.outcome, matchCount: input.matches.length, enteredBy, ageCategorySnapshot: playerAgeCategoryCode },
          outcome: 'success',
        });

        return { ok: true as const, isOverwrite, enteredBy };
      }),
    ```

    ```typescript
    listPendingForPlayer: protectedProcedure
      .input(listPendingForPlayerInput)
      .query(async ({ ctx, input }) => {
        const targetPlayerId = input.playerUserId ?? ctx.scope.userId;
        if (ctx.scope.role === 'player' && targetPlayerId !== ctx.scope.userId) {
          throw new TRPCError({ code: 'FORBIDDEN' });
        }

        const now = new Date();
        const FOURTEEN_DAYS_MS = 14 * 24 * 60 * 60 * 1000;
        const fourteenDaysAgo = new Date(now.getTime() - FOURTEEN_DAYS_MS);

        // Tournaments the player is a participant of, ended in last 14d, NO tournament_results row yet
        const pending = await ctx.db.execute(sql`
          SELECT ce.id AS event_id, ce.starts_at, ce.ends_at, t.naam, t.city, t.country
          FROM calendar_events ce
          JOIN tournaments t ON t.event_id = ce.id
          JOIN calendar_event_participants cep ON cep.event_id = ce.id AND cep.user_id = ${targetPlayerId}
          LEFT JOIN tournament_results tr ON tr.tournament_event_id = ce.id AND tr.player_user_id = ${targetPlayerId}
          WHERE ce.ends_at < ${now}
            AND ce.ends_at >= ${fourteenDaysAgo}
            AND tr.tournament_event_id IS NULL
          ORDER BY ce.ends_at DESC
        `);
        return { pending };
      }),
    ```
