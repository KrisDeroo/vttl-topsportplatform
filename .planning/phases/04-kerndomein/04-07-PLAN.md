    **Step 2 — `src/server/trpc/routers/inbox.ts`:**

    ```typescript
    /**
     * inbox router — Phase 4 minimal stub for D-67 channel 2 + D-72 channel 2.
     *
     * Backed by system_inbox table (Plan 04-02 0020) which is populated nightly
     * at 18:00 Europe/Brussels by pg_cron jobs (Plan 04-02 0019).
     *
     * Phase 6 replaces this with the full Inbox UI (threads / compose / attachments).
     * For Phase 4 we ship 3 read procedures + 1 mark-read mutation.
     *
     * The tone escalation (day 0-6 / 7-9 / 10-12) is purely a UI concern;
     * cron writes the row, client picks the i18n key per maxDaysSinceEnd.
     */
    import { TRPCError } from '@trpc/server';
    import { and, desc, eq, isNull, lt } from 'drizzle-orm';

    import { systemInbox } from '@/server/db/schema';

    import { router } from '../trpc';
    import { writeAudit } from '../middleware/audit';
    import { protectedProcedure } from '../middleware/freshSession';
    import { listInboxInput, markReadInput } from '../schemas/inbox';

    export const inboxRouter = router({
      listUnread: protectedProcedure
        .input(listInboxInput)
        .query(async ({ ctx, input }) => {
          // RLS scopes by user_id = current_user_id(); app layer adds limit + cursor
          const where = [
            eq(systemInbox.userId, ctx.scope.userId),
            isNull(systemInbox.readAt),
          ];
          if (input.cursor) {
            where.push(lt(systemInbox.createdAt, new Date(input.cursor)));
          }

          const rows = await ctx.db.select().from(systemInbox)
            .where(and(...where))
            .orderBy(desc(systemInbox.createdAt))
            .limit(input.limit + 1);

          const hasMore = rows.length > input.limit;
          return {
            entries: rows.slice(0, input.limit),
            nextCursor: hasMore ? rows[input.limit - 1].createdAt.toISOString() : null,
          };
        }),

      listAll: protectedProcedure
        .input(listInboxInput)
        .query(async ({ ctx, input }) => {
          const where = [eq(systemInbox.userId, ctx.scope.userId)];
          if (input.cursor) where.push(lt(systemInbox.createdAt, new Date(input.cursor)));

          const rows = await ctx.db.select().from(systemInbox)
            .where(and(...where))
            .orderBy(desc(systemInbox.createdAt))
            .limit(input.limit + 1);

          const hasMore = rows.length > input.limit;
          return {
            entries: rows.slice(0, input.limit),
            nextCursor: hasMore ? rows[input.limit - 1].createdAt.toISOString() : null,
          };
        }),

      markRead: protectedProcedure
        .input(markReadInput)
        .mutation(async ({ ctx, input }) => {
          // RLS guarantees we only update our own row; SELECT to confirm row exists for NOT_FOUND.
          const existing = await ctx.db.select().from(systemInbox).where(
            and(eq(systemInbox.id, input.id), eq(systemInbox.userId, ctx.scope.userId)),
          ).limit(1);
          if (!existing[0]) throw new TRPCError({ code: 'NOT_FOUND' });

          if (existing[0].readAt !== null) {
            return { ok: true as const, alreadyRead: true };
          }

          await ctx.db.update(systemInbox)
            .set({ readAt: new Date() })
            .where(eq(systemInbox.id, input.id));

          await writeAudit(ctx, {
            action: 'inbox_marked_read',
            resourceType: 'system_inbox',
            resourceId: input.id,
            newValues: { kind: existing[0].kind },
            outcome: 'success',
          });

          return { ok: true as const, alreadyRead: false };
        }),
    });
    ```
