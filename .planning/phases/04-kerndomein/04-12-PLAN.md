/**
 * Playwright e2e: tournament result-route targeting (CR-03 + WARNING-3).
 *
 * Scenarios:
 *   1. TD navigates to /tournaments/[id]/result WITHOUT ?playerId
 *      → page renders Pick-Player selector with the FULL participants
 *        roster (from tournament.get.participants, not just rows with
 *        existing results). Does NOT auto-fill any existing.results[0] data.
 *   2. TD navigates to /tournaments/[id]/result?playerId=PLAYER_B_ID
 *      → form preloads PLAYER_B's existing row (if any), NOT player_A's.
 *   3. Player navigates to /tournaments/[id]/result (no ?playerId)
 *      → form preloads OWN row (player caller defaults to self).
 *   4. NEW (WARNING-3 regression): TD on a tournament with NO entered
 *      results yet sees the Pick-Player selector populated with the
 *      participants list (this case used to render empty when the picker
 *      fell back to existing.results only).
 *
 * Skip-gate: if the test-auth hook is unavailable (no TEST_AUTH_TOKEN
 * env), the spec exits cleanly. Same pattern as rankings-tab.spec.ts.
 *
 * Reference: .planning/phases/04-kerndomein/04-VERIFICATION.md §gaps[3]
 *            .planning/phases/04-kerndomein/04-REVIEW.md §CR-03
 */
import { test, expect } from '@playwright/test';

const TEST_AUTH = process.env.TEST_AUTH_TOKEN;
const TEST_TOURNAMENT_ID = process.env.TEST_TOURNAMENT_ID;
const TEST_PLAYER_B_ID = process.env.TEST_PLAYER_B_ID;
const TEST_EMPTY_TOURNAMENT_ID = process.env.TEST_EMPTY_TOURNAMENT_ID;

test.describe('Tournament result-route targeting (CR-03)', () => {
  test.skip(!TEST_AUTH || !TEST_TOURNAMENT_ID, 'TEST_AUTH_TOKEN + TEST_TOURNAMENT_ID required');

  test('TD without ?playerId → Pick-Player selector', async ({ page, baseURL }) => {
    await page.goto(`${baseURL}/api/test-auth?role=technical_director&token=${TEST_AUTH}`);
    await page.goto(`${baseURL}/nl/tournaments/${TEST_TOURNAMENT_ID}/result`);
    // Picker UI heading or empty-state copy must be visible.
    await expect(page.getByText(/Kies een speler|Pick a player|Choisissez un joueur/)).toBeVisible();
    // Form fields (outcome select, match table) must NOT be present.
    await expect(page.locator('[name=outcome]')).toHaveCount(0);
  });

  test('TD with ?playerId=PLAYER_B → form preloads PLAYER_B row', async ({ page, baseURL }) => {
    test.skip(!TEST_PLAYER_B_ID, 'TEST_PLAYER_B_ID required');
    await page.goto(`${baseURL}/api/test-auth?role=technical_director&token=${TEST_AUTH}`);
    await page.goto(`${baseURL}/nl/tournaments/${TEST_TOURNAMENT_ID}/result?playerId=${TEST_PLAYER_B_ID}`);
    // Form must be present.
    await expect(page.locator('form, [data-testid=tournament-result-form]')).toBeVisible({ timeout: 5_000 });
    // Picker must NOT be present.
    await expect(page.getByText(/Kies een speler|Pick a player/)).toHaveCount(0);
  });

  test('Player without ?playerId → form preloads own row', async ({ page, baseURL }) => {
    await page.goto(`${baseURL}/api/test-auth?role=player&token=${TEST_AUTH}`);
    await page.goto(`${baseURL}/nl/tournaments/${TEST_TOURNAMENT_ID}/result`);
    await expect(page.locator('form, [data-testid=tournament-result-form]')).toBeVisible({ timeout: 5_000 });
  });

  test('TD on empty-results tournament sees full participants in picker (WARNING-3 regression)', async ({ page, baseURL }) => {
    test.skip(!TEST_EMPTY_TOURNAMENT_ID, 'TEST_EMPTY_TOURNAMENT_ID required (a tournament with participants but zero entered results)');
    await page.goto(`${baseURL}/api/test-auth?role=technical_director&token=${TEST_AUTH}`);
    await page.goto(`${baseURL}/nl/tournaments/${TEST_EMPTY_TOURNAMENT_ID}/result`);
    // Picker is visible.
    await expect(page.getByText(/Kies een speler|Pick a player|Choisissez un joueur/)).toBeVisible();
    // Picker has at LEAST one option (the participants list backs it via
    // tournament.get.participants). The original spec (empty-floor fallback)

    await expect(page.getByText(/Geen deelnemers gevonden|No participants found|Aucun participant/)).toHaveCount(0);
    // And at least one link to a ?playerId= URL is present.
    const playerLinks = await page.locator('a[href*="playerId="]').count();
    expect(playerLinks).toBeGreaterThan(0);
  });
});
